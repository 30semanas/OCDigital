Código para o Google Apps Script

No seu Google Sheets, acesse Extensões → Apps Script, cole o código abaixo e publique como App da Web com acesso de Qualquer pessoa. 

function doPost(e) {
  const d = JSON.parse(e.postData.contents);
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  // Aba: OC
  let s = ss.getSheetByName('OC') || ss.insertSheet('OC');
  s.clearContents();
  s.appendRow(['Campus','Data','Tema','GC','Mensagem','Semanas','Aviso Telão']);
  s.appendRow([d.meta.campus,d.meta.data,d.meta.tema,
               d.meta.gc,d.meta.msg,d.meta.semanas,d.meta.aviso]);
  s.appendRow([]);
  s.appendRow(['Duração','Horário','Momento','Responsável','Vídeo/Som',
               'Tipo Momento','Telão Central','Transmissão',
               'Vídeo/Dança/Luz','GC Suspenso']);
  d.rows.forEach(r => s.appendRow([
    r.dur,r.hora,r.momento,r.resp,r.videosom,
    r.tipo_momento,r.telao,r.transm,r.vdl,r.gc_susp
  ]));

  // Aba: Bandas
  let sb = ss.getSheetByName('Bandas') || ss.insertSheet('Bandas');
  sb.clearContents();
  sb.appendRow(['Nome da Banda','Ministro Líder']);
  d.bandas.forEach(b => sb.appendRow([b.nome, b.lider]));

  // Aba: Pessoas
  let sp = ss.getSheetByName('Pessoas') || ss.insertSheet('Pessoas');
  sp.clearContents();
  sp.appendRow(['Nome','Função','Telefone']);
  d.pessoas.forEach(p => sp.appendRow([p.nome, p.func, p.tel]));

  return ContentService.createTextOutput('ok')
    .setMimeType(ContentService.MimeType.TEXT);
}
